Total Sales = SUM(cleaned_sales_data[sales])

Total Profit = SUM(cleaned_sales_data[profit])

Profit Margin % = DIVIDE([Total Profit], [Total Sales], 0)

Total Orders = DISTINCTCOUNT(cleaned_sales_data[order_id])

Average Order Value = DIVIDE([Total Sales], [Total Orders], 0)

Total Customers = DISTINCTCOUNT(cleaned_sales_data[customer_id])

Units Sold = SUM(cleaned_sales_data[quantity])

// Mark order_date as a date table column first (or use a Calendar table), then:
Sales Last Year = CALCULATE([Total Sales], SAMEPERIODLASTYEAR(cleaned_sales_data[order_date]))

YoY Growth % = DIVIDE([Total Sales] - [Sales Last Year], [Sales Last Year], 0)

Sales YTD = TOTALYTD([Total Sales], cleaned_sales_data[order_date])
